Task 2: Adafruit IO MQTT & Soldered Board

Task 2 advances beyond local microcontroller logic to full cloud-connected industrial automation. By integrating the lightweight MQTT (Message Queuing Telemetry Transport) protocol with the Adafruit IO cloud dashboard, the ESP32 receives real-time publish/subscribe packets to switch a high-voltage 230V AC mains bulb via a relay module mounted on a custom soldered prototype board.

Hardware Architecture & MQTT Feed Parameters

Arduino C++ Source Code (Adafruit MQTT Client)

Field-tested firmware using WiFi.h and Adafruit_MQTT_Client.h. Implements auto-reconnect loops and non-blocking feed polling.

ESP32_Adafruit_IO_MQTT_Relay.ino 
/*
 * =====================================================================
 * Task 2: ESP32 + Relay + Adafruit IO (MQTT) — 230V Bulb ON/OFF Control
 * Developer: PRAVEEN. A
 * Board: ESP32 DevKit V1 + Custom Soldered Perfboard
 * Load: 230V AC Bulb via Single-Channel Relay (COM/NO)
 * =====================================================================
 */

#include <WiFi.h>
#include "Adafruit_MQTT.h"
#include "Adafruit_MQTT_Client.h"

// ---------- WiFi Credentials ----------
#define WLAN_SSID       "protosem"
#define WLAN_PASS       "Proto#123"

// ---------- Adafruit IO Credentials ----------
#define AIO_SERVER      "io.adafruit.com"
#define AIO_SERVERPORT  1883
#define AIO_USERNAME    "elaks_22"
#define AIO_KEY         "aio_Hdzy40ALTmyX6Pqhtzw8Qg1fUMiV"

// ---------- Relay Control Pin ----------
#define RELAY_PIN 5

// ---------- WiFi + MQTT Client Setup ----------
WiFiClient client;
Adafruit_MQTT_Client mqtt(&client, AIO_SERVER, AIO_SERVERPORT, AIO_USERNAME, AIO_KEY);

// ---------- Feed Subscription Setup ----------
Adafruit_MQTT_Subscribe bulbFeed = Adafruit_MQTT_Subscribe(&mqtt, AIO_USERNAME "/feeds/bulb-control");

void MQTT_connect();

void setup() {
  Serial.begin(115200);
  delay(100);

  // Configure relay pin
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW);   // Bulb OFF initially

  // Connect to Local WiFi Network
  Serial.print("Connecting to WiFi: ");
  Serial.println(WLAN_SSID);
  WiFi.begin(WLAN_SSID, WLAN_PASS);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi connected! IP Address: ");
  Serial.println(WiFi.localIP());

  // Subscribe to Adafruit IO Feed
  mqtt.subscribe(&bulbFeed);
}

void loop() {
  // Ensure MQTT connection is active
  MQTT_connect();

  // Poll for incoming subscription packets (5-second timeout window)
  Adafruit_MQTT_Subscribe *subscription;
  while ((subscription = mqtt.readSubscription(5000))) {
    if (subscription == &bulbFeed) {
      String value = (char *)bulbFeed.lastread;
      Serial.print("[MQTT INCOMING] Feed Value: ");
      Serial.println(value);

      // Actuate relay based on cloud command
      if (value == "1" || value == "ON" || value == "on") {
        digitalWrite(RELAY_PIN, HIGH);   // Relay ON -> Bulb ON
        Serial.println("[ACTION] Relay Triggered -> 230V BULB ON");
      } else if (value == "0" || value == "OFF" || value == "off") {
        digitalWrite(RELAY_PIN, LOW);    // Relay OFF -> Bulb OFF
        Serial.println("[ACTION] Relay Released -> 230V BULB OFF");
      }
    }
  }

  // Ping broker to maintain keep-alive heartbeat
  if (!mqtt.ping()) {
    mqtt.disconnect();
  }
}

// Function to handle robust MQTT reconnection
void MQTT_connect() {
  int8_t ret;
  if (mqtt.connected()) {
    return;
  }

  Serial.print("Connecting to Adafruit IO MQTT... ");
  uint8_t retries = 3;
  while ((ret = mqtt.connect()) != 0) {
    Serial.println(mqtt.connectErrorString(ret));
    Serial.println("Retrying connection in 5 seconds...");
    mqtt.disconnect();
    delay(5000);
    retries--;
    if (retries == 0) {
      Serial.println("[FATAL] MQTT Connection failed. Resetting loop.");
      return;
    }
  }
  Serial.println("Adafruit IO Connected!");
}
